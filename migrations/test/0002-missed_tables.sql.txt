-- Bit X/18: students
-- ============================================

CREATE TABLE IF NOT EXISTS students (
    id BIGSERIAL PRIMARY KEY,
    school_id BIGINT NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    user_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    admission_no VARCHAR(50),
    enrollment_date DATE DEFAULT NOW(),
    guardian_contact VARCHAR(100),
    status VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMPTZ DEFAULT NOW(),
    created_by BIGINT,
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    updated_by BIGINT,
    is_deleted BOOLEAN DEFAULT FALSE,
    deleted_at TIMESTAMPTZ,
    deleted_by BIGINT,
    UNIQUE(school_id, admission_no),
    UNIQUE(school_id, user_id)
);

ALTER TABLE students ENABLE ROW LEVEL SECURITY;

DO $$BEGIN
    IF EXISTS (SELECT 1 FROM pg_policies WHERE tablename='students' AND policyname='students_isolation') THEN
        EXECUTE 'DROP POLICY students_isolation ON students';
    END IF;
END$$;

CREATE POLICY students_isolation ON students
    FOR ALL TO authenticated
    USING (school_id = current_tenant_id() AND NOT is_deleted)
    WITH CHECK (school_id = current_tenant_id() AND NOT is_deleted);

CREATE INDEX IF NOT EXISTS idx_students_school_user ON students(school_id, user_id);

DROP TRIGGER IF EXISTS trg_students_updated_at ON students;
CREATE TRIGGER trg_students_updated_at
BEFORE UPDATE ON students
FOR EACH ROW EXECUTE FUNCTION set_updated_at();

-- CRUD Functions
CREATE OR REPLACE FUNCTION insert_student(
    _school_id BIGINT,_user_id BIGINT,_admission_no TEXT,_guardian_contact TEXT,_status TEXT
) RETURNS BIGINT AS $$
DECLARE new_id BIGINT;
BEGIN
    INSERT INTO students (school_id,user_id,admission_no,guardian_contact,status,created_by)
    VALUES (_school_id,_user_id,_admission_no,_guardian_contact,COALESCE(_status,'active'),current_user_id())
    RETURNING id INTO new_id;
    RETURN new_id;
END; $$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION update_student(
    _id BIGINT,_school_id BIGINT,_admission_no TEXT,_guardian_contact TEXT,_status TEXT
) RETURNS VOID AS $$
BEGIN
    UPDATE students SET
        admission_no=COALESCE(_admission_no,admission_no),
        guardian_contact=COALESCE(_guardian_contact,guardian_contact),
        status=COALESCE(_status,status),
        updated_by=current_user_id()
    WHERE id=_id AND school_id=_school_id AND is_deleted=FALSE;
END; $$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION soft_delete_student(
    _id BIGINT,_school_id BIGINT
) RETURNS VOID AS $$
BEGIN
    UPDATE students SET is_deleted=TRUE,deleted_at=NOW(),deleted_by=current_user_id()
    WHERE id=_id AND school_id=_school_id AND is_deleted=FALSE;
END; $$ LANGUAGE plpgsql SECURITY DEFINER;

-- Grants
GRANT SELECT,INSERT,UPDATE ON students TO authenticated;
GRANT EXECUTE ON FUNCTION insert_student TO authenticated;
GRANT EXECUTE ON FUNCTION update_student TO authenticated;
GRANT EXECUTE ON FUNCTION soft_delete_student TO authenticated;

-- 📊 staff_promotion_history_view

DROP VIEW IF EXISTS staff_promotion_history_view;

CREATE VIEW staff_promotion_history_view AS
SELECT
    sp.id AS promotion_id,
    sp.school_id,
    st.id AS staff_id,
    (u.first_name || ' ' || u.last_name) AS staff_name,
    st.employee_no,
    st.department,
    sp.from_role,
    sp.to_role,
    sp.from_department,
    sp.to_department,
    sp.promotion_date,
    sp.status AS promotion_status,
    sp.remarks,
    sp.created_at,
    sp.updated_at
FROM staff_promotions sp
JOIN staff st ON sp.staff_id = st.id
JOIN users u ON st.user_id = u.id
WHERE sp.is_deleted = FALSE
  AND st.is_deleted = FALSE
  AND u.is_deleted = FALSE
ORDER BY sp.promotion_date DESC;


-- 📊 student_promotion_history_view

DROP VIEW IF EXISTS student_promotion_history_view;

CREATE VIEW student_promotion_history_view AS
SELECT
    sp.id AS promotion_id,
    sp.school_id,
    st.id AS student_id,
    (u.first_name || ' ' || u.last_name) AS student_name,
    st.admission_no,
    sp.from_class_id,
    c_from.name AS from_class_name,
    sp.to_class_id,
    c_to.name AS to_class_name,
    sp.from_grade_level_id,
    gl_from.name AS from_grade_level,
    sp.to_grade_level_id,
    gl_to.name AS to_grade_level,
    sp.academic_year_id,
    ay.name AS academic_year,
    sp.promotion_date,
    sp.status AS promotion_status,
    sp.remarks,
    sp.created_at,
    sp.updated_at
FROM student_promotions sp
JOIN students st ON sp.student_id = st.id
JOIN users u ON st.user_id = u.id
JOIN classes c_from ON sp.from_class_id = c_from.id
JOIN classes c_to ON sp.to_class_id = c_to.id
JOIN grade_levels gl_from ON sp.from_grade_level_id = gl_from.id
JOIN grade_levels gl_to ON sp.to_grade_level_id = gl_to.id
JOIN academic_years ay ON sp.academic_year_id = ay.id
WHERE sp.is_deleted = FALSE
  AND st.is_deleted = FALSE
  AND u.is_deleted = FALSE
ORDER BY sp.promotion_date DESC;


-- 📊 promotion_summary_view (with percentages)

DROP VIEW IF EXISTS promotion_summary_view;

CREATE VIEW promotion_summary_view AS
WITH student_summary AS (
    SELECT
        sp.school_id,
        sp.academic_year_id,
        ay.name AS academic_year,
        COUNT(*) FILTER (WHERE sp.status = 'promoted') AS students_promoted,
        COUNT(*) FILTER (WHERE sp.status = 'repeated') AS students_repeated,
        COUNT(*) FILTER (WHERE sp.status = 'transferred') AS students_transferred,
        COUNT(*) AS total_students_considered
    FROM student_promotions sp
    JOIN academic_years ay ON sp.academic_year_id = ay.id
    WHERE sp.is_deleted = FALSE
    GROUP BY sp.school_id, sp.academic_year_id, ay.name
),
staff_summary AS (
    SELECT
        sp.school_id,
        EXTRACT(YEAR FROM sp.promotion_date)::INT AS academic_year_id,
        COUNT(*) FILTER (WHERE sp.status = 'promoted') AS staff_promoted,
        COUNT(*) FILTER (WHERE sp.status = 'transferred') AS staff_transferred,
        COUNT(*) FILTER (WHERE sp.status = 'acting') AS staff_acting,
        COUNT(*) AS total_staff_considered
    FROM staff_promotions sp
    WHERE sp.is_deleted = FALSE
    GROUP BY sp.school_id, EXTRACT(YEAR FROM sp.promotion_date)
)
SELECT
    COALESCE(st.school_id, ss.school_id) AS school_id,
    COALESCE(st.academic_year_id, ss.academic_year_id) AS academic_year_id,
    COALESCE(st.academic_year, ss.academic_year_id::TEXT) AS academic_year,
    -- Student counts
    COALESCE(st.students_promoted,0) AS students_promoted,
    COALESCE(st.students_repeated,0) AS students_repeated,
    COALESCE(st.students_transferred,0) AS students_transferred,
    COALESCE(st.total_students_considered,0) AS total_students_considered,
    -- Student percentages
    ROUND(COALESCE(st.students_promoted,0)::NUMERIC / NULLIF(st.total_students_considered,0) * 100,2) AS pct_students_promoted,
    ROUND(COALESCE(st.students_repeated,0)::NUMERIC / NULLIF(st.total_students_considered,0) * 100,2) AS pct_students_repeated,
    ROUND(COALESCE(st.students_transferred,0)::NUMERIC / NULLIF(st.total_students_considered,0) * 100,2) AS pct_students_transferred,
    -- Staff counts
    COALESCE(ss.staff_promoted,0) AS staff_promoted,
    COALESCE(ss.staff_transferred,0) AS staff_transferred,
    COALESCE(ss.staff_acting,0) AS staff_acting,
    COALESCE(ss.total_staff_considered,0) AS total_staff_considered,
    -- Staff percentages
    ROUND(COALESCE(ss.staff_promoted,0)::NUMERIC / NULLIF(ss.total_staff_considered,0) * 100,2) AS pct_staff_promoted,
    ROUND(COALESCE(ss.staff_transferred,0)::NUMERIC / NULLIF(ss.total_staff_considered,0) * 100,2) AS pct_staff_transferred,
    ROUND(COALESCE(ss.staff_acting,0)::NUMERIC / NULLIF(ss.total_staff_considered,0) * 100,2) AS pct_staff_acting
FROM student_summary st
FULL OUTER JOIN staff_summary ss
  ON st.school_id = ss.school_id
 AND st.academic_year_id = ss.academic_year_id
ORDER BY academic_year_id DESC;


-- ============================================
-- Helper Functions
-- ============================================

CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION current_user_id()
RETURNS BIGINT AS $$
BEGIN
  RETURN (auth.uid())::BIGINT;
END;
$$ LANGUAGE plpgsql STABLE;

CREATE OR REPLACE FUNCTION current_school_id()
RETURNS BIGINT AS $$
BEGIN
  RETURN (auth.jwt() ->> 'school_id')::BIGINT;
END;
$$ LANGUAGE plpgsql STABLE;

CREATE OR REPLACE FUNCTION log_audit()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO audit_logs (
    school_id, user_id, action, table_name, record_id,
    old_values, new_values, created_by
  )
  VALUES (
    current_school_id(),
    current_user_id(),
    TG_OP,
    TG_TABLE_NAME,
    COALESCE(NEW.id, OLD.id),
    to_jsonb(OLD),
    to_jsonb(NEW),
    current_user_id()
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================
-- Admission Helper Function
-- ============================================

CREATE OR REPLACE FUNCTION admit_student_if_passed(
    _student_id BIGINT,
    _assessment_id BIGINT,
    _pass_mark NUMERIC
) RETURNS BOOLEAN AS $$
DECLARE
    result_score NUMERIC;
BEGIN
    SELECT score INTO result_score
    FROM assessment_results
    WHERE student_id = _student_id
      AND assessment_id = _assessment_id
      AND is_deleted = FALSE
    LIMIT 1;

    IF result_score IS NULL THEN
        RETURN FALSE;
    END IF;

    IF result_score >= _pass_mark THEN
        UPDATE students
        SET admission_date = NOW(),
            admission_status_id = (SELECT id FROM statuses WHERE label = 'admitted' AND module = 'admission' LIMIT 1),
            updated_by = current_user_id()
        WHERE id = _student_id AND is_deleted = FALSE;
        RETURN TRUE;
    ELSE
        UPDATE students
        SET admission_status_id = (SELECT id FROM statuses WHERE label = 'rejected' AND module = 'admission' LIMIT 1),
            updated_by = current_user_id()
        WHERE id = _student_id AND is_deleted = FALSE;
        RETURN FALSE;
    END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

GRANT EXECUTE ON FUNCTION admit_student_if_passed(BIGINT,BIGINT,NUMERIC) TO authenticated;
